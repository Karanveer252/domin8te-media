<IfModule mod_headers.c>
  # Site wide, safe for every page (Meta Pixel, Calendly and Clerk are not affected).
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>

# The console and the dashboard may only be framed by our own pages (the console's view-as shows the
# dashboard inside the console, same origin). This stops other sites from framing them to trick a click.
# The public demo dashboard (/dashboard/demo/) is left out on purpose in case it should stay embeddable;
# remove the second condition in the <If> below if it should be locked too.
<IfModule mod_headers.c>
  <If "%{REQUEST_URI} =~ m#^/(console|dashboard)(/|$)# && %{REQUEST_URI} !~ m#^/dashboard/demo(/|$)#">
    Header always set Content-Security-Policy "frame-ancestors 'self'"
    Header always set X-Frame-Options "SAMEORIGIN"
  </If>
</IfModule>
